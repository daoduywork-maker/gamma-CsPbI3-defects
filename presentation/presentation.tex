\section{Motivation}

\begin{frame}{Why all-inorganic CsPbI$_3$}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \small
      \begin{itemize}
        \item \textbf{Record cells are hybrid.} FA-rich mixed-cation, mixed-halide perovskites reached about 28\,\% (single junction)\cit{1}.
        \item \textbf{Organic cations limit thermal stability.} MA$^+$ and FA$^+$ are volatile at processing temperatures.
        \item \textbf{All-inorganic CsPbI$_3$} has no volatile cation and a gap of about 1.7 eV, suited to tandem top cells. Best cells reach about 21--22\,\%.
        \item \textbf{The obstacle is the phase}, not the absorber itself.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \scriptsize
      \setlength{\tabcolsep}{4pt}
      \begin{tabular}{lcc}
        \toprule
        & Hybrid & CsPbI$_3$ \\
        \midrule
        Best efficiency & $\approx$28\,\% & $\approx$21--22\,\% \\
        Volatile cation & yes & no \\
        Black phase stable at RT & mostly & \textbf{no} \\
        \bottomrule
      \end{tabular}
    \end{column}
  \end{columns}
  \refs{\RA\ Values approximate; records change often.}
\end{frame}

\begin{frame}{The phase problem}
  \small
  \begin{itemize}
    \item \textbf{Four phases.} Cubic $\alpha$ (above about 300\,$^\circ$C) $\rightarrow$ tetragonal $\beta$ $\rightarrow$ orthorhombic \textbf{$\gamma$} on cooling, by octahedral tilting. All three are black (photoactive).
    \item \textbf{$\delta$ is the ground state} at room temperature: yellow, edge-sharing PbI$_6$ chains, gap about 2.8 eV\cit{2}.
    \item \textbf{$\gamma$ is metastable with respect to $\delta$}, not to $\alpha$. Quenching keeps $\gamma$ because $\alpha\rightarrow\beta\rightarrow\gamma$ is displacive and fast, whereas $\gamma\rightarrow\delta$ rebuilds the network and is slow.
    \item \textbf{This work (PBEsol, 0 K):} $\delta$ lies \alert{79 meV per formula unit} below $\gamma$ (70 meV with SOC) and is 6\,\% denser (218.9 vs 232.3 \AA$^3$ per formula unit).
  \end{itemize}
  \refs{\RB}
\end{frame}

\begin{frame}{Strain: a stabiliser that also moves the ions}
  \small
  \begin{itemize}
    \item \textbf{Clamping keeps the black phase.} Films quenched from 330\,$^\circ$C on glass or ITO kept about 1\,\% in-plane tensile strain and stayed black\cit{3,4}. Lattice-mismatched nanoplates show the same trend\cit{16}.
    \item \textbf{Tensile strain lowers migration barriers.} In CsPbI$_2$Br the barrier fell from 0.67 to 0.55 eV at +1.5\,\% and rose to 0.79 eV at $-1.5$\,\%\cit{5}.
    \item \textbf{The response is hop-resolved.} In Pnma CsPbBr$_3$ the diffusivity tensor changed differently along each axis, and not monotonically\cit{6}.
    \item \textbf{Bulk \cpi\ under strain}\cit{7}: two migration paths only, charge state not stated.
  \end{itemize}
  \begin{alertblock}{Tension}
    \small Tensile strain helps the phase but speeds up ion migration. No study has put both on one strain scale for \cpi.
  \end{alertblock}
  \refs{\RC\quad \RD\quad \RE\quad \RF\quad \RG\quad \RP}
\end{frame}

\begin{frame}{Research questions}
  \small
  \begin{enumerate}
    \item \textbf{Phase.} How biaxial (001) strain changes the $\gamma$--$\delta$ energy difference.
    \item \textbf{Formation.} How strain changes the formation energy of $V_\mathrm{I}$, and which charge state is relevant across the gap.
    \item \textbf{Migration.} Which hops carry the transport, and how strain changes the barriers and the diffusivity tensor.
    \item \textbf{Charge state.} Whether the neutral vacancy exists in the gap, and whether its barrier differs from $+1$.
  \end{enumerate}
  \vspace{0.5em}
  \textbf{Strains:} $\varepsilon = -1,\ -0.5,\ 0,\ +0.5,\ +1$\,\% on $a$ and $b$, with $c$ relaxed. The tensile end matches the clamping strain of Steele \emph{et al.}\cit{3}
  \refs{\RC}
\end{frame}

\section{Results: the unstrained crystal}

\begin{frame}{Site energies and the charge state}
  \centering\small
  \begin{tabular}{lccc}
    \toprule
    $E(\mathrm{8d}) - E(\mathrm{4c})$ & PBEsol & PBEsol + SOC (single point) & PBEsol + SOC (relaxed) \\
    \midrule
    $V_\mathrm{I}^{+}$ & +0.9 meV & +1.0 meV & --- \\
    $V_\mathrm{I}^{0}$ & +33.5 meV & $-71$ meV & $\mathbf{-1.4}$ \textbf{meV} \\
    \bottomrule
  \end{tabular}

  \vspace{0.6em}
  \raggedright
  \begin{itemize}
    \item \textbf{$+1$:} the two sites are degenerate, and SOC changes nothing. The Pb-6p conduction band is empty.
    \item \textbf{Neutral:} the extra electron enters the Pb-6p conduction band, spread over the cell (shallow donor, magnetisation 0). SOC shifts that band by about 1 eV, so the neutral site energies depend on the method.
    \item \textbf{Consequence:} $+1$ results are robust at PBEsol. Neutral results need SOC (and possibly a hybrid functional).
  \end{itemize}
\end{frame}

\begin{frame}{Diffusivity tensor of $V_\mathrm{I}^{+}$}
  \small

  \vspace{0.3em}

        $b$ & $2.6\times10^{-9}$ & 0.332 eV \\
        $c$ & $4.1\times10^{-9}$ & 0.338 eV \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering\scriptsize
      \begin{tabular}{lccc}
        \toprule
        Hop removed & $D_{aa}$ & $D_{bb}$ & $D_{cc}$ \\
        \midrule
        S1 & $\times$0.40 & $\times$0.92 & $\times$0.76 \\
        S2 & $\times$0.99 & $\times$\textbf{0.04} & $\times$0.72 \\
        S4 & $\times$0.63 & $\times$0.35 & $\times$0.72 \\

        \bottomrule
      \end{tabular}
    \end{column}

  \vspace{0.4em}
  \begin{itemize}
    \item \textbf{Nearly isotropic.} Shifting the four fast barriers by $\pm$0.02 eV changes $D_{cc}/D_{aa}$ from 0.5 to 3.0, so the axis order is not resolved.
    \item \textbf{Robust:} each axis runs on a different hop ($b$ almost only on S2). Strain that shifts S1 and S2 differently changes the anisotropy.
  \end{itemize}
  \refs{\RF}
\end{frame}

\begin{frame}{Points for discussion}
  \small
  \begin{enumerate}
    \item \textbf{Strained NEBs.} Four fast hops (S2, S4, S1, S5) at four strains make 16 NEBs of 2--4 days on 4 nodes each. S2 and S4 alone carry most of the transport.
    \item \textbf{Charge state.} $V_\mathrm{I}$ is $+1$ throughout the gap (HSE06 + SOC), so the charge-state question is answered without a neutral NEB.
    \item \textbf{Phase result.} Strain does not stabilise $\gamma$ against $\delta$ at $\pm$1\,\%; both limits reported, with the kinetic reading.
    \item \textbf{Prefactors.} Kept at $10^{12}$ Hz; the axis order of the tensor stays open.
    \item \textbf{Fermi level.} A self-consistent $E_F$ needs acceptors ($V_\mathrm{Pb}$, $V_\mathrm{Cs}$), outside the present scope.
  \end{enumerate}
\end{frame}

\begin{frame}{Summary}
  \small
  \begin{itemize}
    \item \textbf{$V_\mathrm{I}$ is $+1$ for every Fermi level in the gap} (HSE06 + SOC edges). Its energies and barriers are robust at PBEsol.
    \item \textbf{Four short hops within 0.02 eV} (0.326--0.348 eV) carry the transport, each axis on a different hop. The tensor is nearly isotropic.
    \item \textbf{Strain does not stabilise $\gamma$}: the $\gamma$--$\delta$ gap (79 meV/f.u.) is smallest at zero strain and grows by up to 5 meV at $\pm$1\,\%. Clamping must act kinetically.
    \item \textbf{Tension lowers the vacancy formation energy} by about 0.034 eV per \%: tensile films hold more iodine vacancies.
  \end{itemize}
