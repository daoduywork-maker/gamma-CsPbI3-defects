\begin{frame}{The phase problem}
  % Pls mark where is alpha, beta, gamma, delta
  \begin{columns}[c]
    \begin{column}{0.56\textwidth}
      \footnotesize
      \begin{itemize}
        \item \textbf{Four phases.} Cubic $\alpha$ $\rightarrow$ tetragonal $\beta$ $\rightarrow$ orthorhombic \textbf{$\gamma$} on cooling, by octahedral tilting. All three are black (photoactive).
        \item \textbf{$\delta$ is the ground state} at room temperature. It is yellow, with edge-sharing PbI$_6$ chains and a gap of about 2.8 eV\cit{Sutton 2018}.
        \item \textbf{Slow cooling gives $\delta$.} $\alpha$ is stable only between 599 and 758 K (yellow box), and $\delta$ below 599 K\cit{van Hattem 2023}.
        \item \textbf{Quenching is easy. Keeping $\gamma$ is the hard part.} Fast cooling turns $\alpha$ into $\gamma$ by tilting, a displacive and fast change. At room temperature $\gamma$ then turns into $\delta$, quickly in humid air\cit{Steele 2019}. This step rebuilds the network and is slow, but not slow enough.
        \item \alert{Stabilising $\gamma$ means slowing $\gamma\rightarrow\delta$}, not forming $\gamma$.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.68\textheight,keepaspectratio]{phase_CsI_PbI2_film.png}

      \tiny CsI--PbI$_2$ phase diagram, data of van Hattem \emph{et al.} (2023). $\beta$ and $\gamma$ are absent because they are never stable.
    \end{column}
  \end{columns}
  \refs{\RB\quad \RC\quad \RT}
\end{frame}

\begin{frame}{Strain affects the phase and ion migration in different ways}
  \footnotesize
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \textbf{Clamping keeps the black phase.} Films quenched from 330\,$^\circ$C on glass or ITO kept about 1\,\% in-plane tensile strain and stayed black\cit{Steele 2019, Steele 2021}. Nanoplates on mica agreed\cit{Le 2023}.
    \item \textbf{Tensile strain tends to lower migration barriers.} In CsPbI$_2$Br the barrier fell from 0.67 to 0.55 eV at +1.5\,\% and rose to 0.79 eV at $-1.5$\,\%\cit{Xue 2020}.
    \item \textbf{Compression slows ions (group work).} In Cs$_2$AgInBr$_6$ a compressive biaxial strain of only 0.5\,\% raised the Ag$_i^+$ barrier from 0.15 to 0.24 eV\cit{Xu 2019}.
    \item \textbf{The response is hop-resolved.} In Pnma CsPbBr$_3$ the diffusivity tensor changed differently along each axis, and not monotonically\cit{Smolders 2024}.
    \item \textbf{Bulk \cpi\ under strain}\cit{Zhang 2025}. Two migration paths, neutral vacancy by default. HSE06 + SOC (this work) leaves $+1$ as the only stable charge state in the gap.
  \end{itemize}
  \vspace{0.2em}
  \alert{\textbf{Overall.} Tensile strain helps the phase but speeds up ion migration. No study has put both on one strain scale for \cpi.}
  \refs{\RC\quad \RD\quad \RP\quad \RE\quad \RJ\quad \RF\quad \RG}
\end{frame}

\begin{frame}{What has not been answered}
  \small

  \begin{enumerate}
    \item \textbf{Phase.} How biaxial strain in the (001) plane changes the $\gamma$--$\delta$ energy difference.
    \item \textbf{Formation and charge state.} How strain changes the formation energy of $V_\mathrm{I}$, and which charge state is stable across the gap.
    \item \textbf{Migration.} Which hops carry the transport, and how strain changes the barriers and the diffusivity tensor.
  \end{enumerate}
  \begin{block}{Why biaxial strain in the (001) plane}
    \footnotesize
    \begin{itemize}
      \item \textbf{Biaxial.} A substrate clamps a film in both in-plane directions, and the film normal relaxes. Thermal-expansion mismatch on cooling therefore gives biaxial strain\cit{Steele 2019, Xue 2020}.
      \item \textbf{(001).} The $ab$ plane of the Pnma cell, with the long $c$ axis free. Zhang \emph{et al.} (2025) found (001) strain the most favourable for keeping the black phase.
      \item \textbf{Open point.} Steele \emph{et al.} (2019) found the long $b$ and $c$ axes in-plane in their films, which is the (100) plane. A (100) check would match those films.
    \end{itemize}
  \end{block}

\end{frame}

\section{Results: the unstrained crystal}

\begin{frame}{Site energies and the charge state}
  \centering\small
  \begin{tabular}{lccc}
    \toprule
    $E(\mathrm{8d}) - E(\mathrm{4c})$ & PBEsol & PBEsol + SOC (single point) & PBEsol + SOC (relaxed) \\
    \midrule
    $V_\mathrm{I}^{+}$ & +0.9 meV & +1.0 meV & --- \\
    $V_\mathrm{I}^{0}$ & +33.5 meV & $-71$ meV & $\mathbf{-1.4}$ \textbf{meV} \\
    \bottomrule
  \end{tabular}

  \vspace{0.6em}
  \raggedright
  \begin{itemize}
    \item \textbf{$+1$:} the two sites are degenerate, and SOC changes nothing. The Pb-6p conduction band is empty.
    \item \textbf{Neutral:} the extra electron enters the Pb-6p conduction band, spread over the cell (shallow donor, magnetisation 0). SOC shifts that band by about 1 eV, so the neutral site energies depend on the method.
    \item \textbf{Consequence:} $+1$ results are robust at PBEsol. Neutral results need SOC (and possibly a hybrid functional).
  \end{itemize}
\end{frame}

\begin{frame}{Points for discussion}
  \small
  \begin{enumerate}
    \item \textbf{Strained NEBs.} Four fast hops (S2, S4, S1, S5) at four strains make 16 NEBs of 2--4 days on 4 nodes each. S2 and S4 alone carry most of the transport.
    \item \textbf{Charge state.} $V_\mathrm{I}$ is $+1$ throughout the gap (HSE06 + SOC), so the charge-state question is answered without a neutral NEB.
    \item \textbf{Phase result.} Strain does not stabilise $\gamma$ against $\delta$ at $\pm$1\,\%; both limits reported, with the kinetic reading.
    \item \textbf{Contrast with Steele \emph{et al.} (2019).} Their DFT found strain destabilised $\delta$ strongly against $\gamma$. How $\delta$ is strained (its own lattice here, the film interface there) needs a check.
    \item \textbf{Prefactors.} Kept at $10^{12}$ Hz; the axis order of the tensor stays open.
    \item \textbf{Fermi level.} A self-consistent $E_F$ needs acceptors ($V_\mathrm{Pb}$, $V_\mathrm{Cs}$), outside the present scope.
  \end{enumerate}
\end{frame}

\begin{frame}{Summary}
  \small
  \begin{itemize}
    \item \textbf{$V_\mathrm{I}$ is $+1$ for every Fermi level in the gap} (HSE06 + SOC edges). Its energies and barriers are robust at PBEsol.
    \item \textbf{Four short hops within 0.02 eV} (0.326--0.348 eV) carry the transport, each axis on a different hop. The tensor is nearly isotropic.
    \item \textbf{Strain does not stabilise $\gamma$}: the $\gamma$--$\delta$ gap (79 meV/f.u.) is smallest at zero strain and grows by up to 5 meV at $\pm$1\,\%. Clamping must act kinetically.
    \item \textbf{Tension lowers the vacancy formation energy} by about 0.034 eV per \%: tensile films hold more iodine vacancies.
  \end{itemize}
  \vspace{0.6em}
  \centering
  \textbf{Next: $-0.5$\,\% supercells, then strained NEBs (does tension also lower the barriers?).}
\end{frame}
