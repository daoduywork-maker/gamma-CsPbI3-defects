\section{Motivation}

\begin{frame}{Why all-inorganic CsPbI$_3$}
  \begin{columns}[c]
    \begin{column}{0.47\textwidth}
      \footnotesize
      \begin{itemize}
        \item \textbf{Record cells are hybrid.} Formamidinium-rich mixed-cation, mixed-halide perovskites reached about 27--28\,\% (single junction)\cit{NREL 2026}.
        \item \textbf{Organic cations limit thermal stability.} Methylammonium and formamidinium are volatile at processing temperatures.

      \end{itemize}

\begin{frame}{The phase problem}
  \begin{columns}[c]
    \begin{column}{0.56\textwidth}
      \footnotesize
      \begin{itemize}
        \item \textbf{Four phases.} On cooling, cubic $\alpha$ $\rightarrow$ tetragonal $\beta$ (533 K) $\rightarrow$ orthorhombic \textbf{$\gamma$} (448 K) by tilting\cit{Marronnier 2018}. All three are black.
        \item \textbf{$\delta$ is the ground state} at room temperature, yellow, with a gap of about 2.8 eV\cit{Sutton 2018}.
        \item \textbf{Slow cooling gives $\delta$.} $\alpha$ is stable only between 599 and 758 K (yellow box), and $\delta$ below 599 K\cit{van Hattem 2023}.
        \item \textbf{Quenching is easy. Keeping $\gamma$ is the hard part.} Fast cooling keeps $\gamma$, but at room temperature $\gamma$ slowly turns into $\delta$, quickly in humid air\cit{Steele 2019}.
        \item \alert{Stabilising $\gamma$ means slowing $\gamma\rightarrow\delta$}, not forming $\gamma$.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.7\textheight,keepaspectratio]{phase_CsI_PbI2_phases.png}

      \tiny CsI--PbI$_2$ phase diagram\cit{van Hattem 2023}. The dashed path marks the metastable $\beta$ and $\gamma$ on fast cooling\cit{Marronnier 2018}.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Strain affects the phase and ion migration in different ways}
  \footnotesize
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \textbf{Clamping keeps the black phase.} Films quenched from 330\,$^\circ$C on glass or ITO kept about 1\,\% in-plane tensile strain and stayed black\cit{Steele 2019, Steele 2021}. Nanoplates on mica agreed\cit{Le 2023}.
    \item \textbf{Tensile strain tends to lower migration barriers.} In CsPbI$_2$Br the barrier fell from 0.67 to 0.55 eV at +1.5\,\% and rose to 0.79 eV at $-1.5$\,\%\cit{Xue 2020}.
    \item \textbf{Compression slows ions (group work).} In Cs$_2$AgInBr$_6$ a compressive biaxial strain of only 0.5\,\% raised the Ag$_i^+$ barrier from 0.15 to 0.24 eV\cit{Xu 2019}.
    \item \textbf{The response is hop-resolved.} In Pnma CsPbBr$_3$ the diffusivity tensor changed differently along each axis, and not monotonically\cit{Smolders 2024}.
    \item \textbf{Bulk \cpi\ under strain}\cit{Zhang 2025}. Two migration paths, neutral vacancy by default. HSE06 + SOC (this work) leaves $+1$ as the only stable charge state in the gap.
  \end{itemize}
  \vspace{0.2em}
  \alert{\textbf{Overall.} Tensile strain helps the phase but speeds up ion migration. No study has put both on one strain scale for \cpi.}
\end{frame}

\begin{frame}{Question 1: does strain change the $\gamma$--$\delta$ balance?}
  \footnotesize
  \textbf{Question.} How biaxial strain in the (001) plane changes the $\gamma$--$\delta$ energy difference.

  \vspace{0.3em}
  \textbf{To calculate.}
  \begin{itemize}
    \item $\gamma$ at each strain: $a = a_0(1+\varepsilon)$, $b = b_0(1+\varepsilon)$, $c$ from a cubic fit of $E(c)$\cit{Xu 2019}.
    \item $\delta$ at the same strain, strained on its own planes (coherent) and relaxed (incoherent).
    \item Energy difference and strain energy per formula unit,
    \[ \Delta E_{\gamma\delta}(\varepsilon) = \frac{E_\gamma(\varepsilon)}{N_\gamma} - \frac{E_\delta(\varepsilon)}{N_\delta},\qquad E_s(\varepsilon) = \frac{E(\varepsilon) - E(0)}{N}. \]
  \end{itemize}
  \begin{block}{Why biaxial strain in the (001) plane}
    \scriptsize
    \begin{itemize}
      \item \textbf{Biaxial.} A substrate clamps a film in both in-plane directions, and the film normal relaxes\cit{Steele 2019}.
      \item \textbf{(001).} The $ab$ plane of the Pnma cell, with the long $c$ axis free, as in Zhang \emph{et al.} (2025), who found (001) strain the most favourable for the black phase.
      \item \textbf{Open point.} Films keep the long $b$ and $c$ axes in-plane, the (100) plane\cit{Steele 2019}.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Question 2: formation energy and charge state}
  \footnotesize
  \textbf{Question.} How strain changes the formation energy of $V_\mathrm{I}$, and which charge state is stable across the gap.

  \vspace{0.3em}
  \textbf{To calculate.} Perfect and defect supercells ($q = +1, 0$) at each strain, band edges (HSE06 + SOC), reference phases.
  \begin{itemize}
    \item Formation energy\cit{Freysoldt 2014, Xu 2017}
    \[ E_f(V_\mathrm{I}^{q}) = E_\mathrm{def}^{q} - E_\mathrm{bulk} + \mu_\mathrm{I} + q\,(E_\mathrm{VBM} + \Delta V + E_F) + E_\mathrm{corr}, \qquad \mu_\mathrm{I} = \mu_\mathrm{I}^{0} + \Delta\mu_\mathrm{I}. \]
    \item Limits of $\Delta\mu$ from the stability of the host against its competing phases
    \[ \Delta\mu_\mathrm{Cs} + \Delta\mu_\mathrm{Pb} + 3\Delta\mu_\mathrm{I} = \Delta H_f(\mathrm{CsPbI_3}),\quad \Delta\mu_\mathrm{Cs} + \Delta\mu_\mathrm{I} < \Delta H_f(\mathrm{CsI}),\quad \Delta\mu_\mathrm{Pb} + 2\Delta\mu_\mathrm{I} < \Delta H_f(\mathrm{PbI_2}). \]
    \item Charge transition level, the Fermi level where two charge states have equal $E_f$
    \[ \varepsilon(q/q') = \frac{E_f(q;\,E_F = 0) - E_f(q';\,E_F = 0)}{q' - q}. \]
  \end{itemize}
\end{frame}

\begin{frame}{Question 3: migration under strain}
  \footnotesize
  \textbf{Question.} Which hops carry the transport, and how strain changes the barriers and the diffusivity tensor.

  \vspace{0.3em}
  \textbf{To calculate.} CI-NEB for every distinct hop, first unstrained, then the fast hops at each strain.
  \begin{itemize}
    \item Barrier from the climbing image\cit{Henkelman 2000}
    \[ E_a = E_\mathrm{saddle} - E_\mathrm{initial}. \]
    \item Hop rate from harmonic transition-state theory\cit{Vineyard 1957}
    \[ \Gamma = \nu^{*} \exp\!\left(-\frac{E_a}{k_B T}\right),\qquad \nu^{*} = \frac{\prod_{i=1}^{3N}\nu_i^\mathrm{initial}}{\prod_{j=1}^{3N-1}\nu_j^\mathrm{saddle}} \approx 10^{12}\ \mathrm{Hz}. \]
    \item Diffusivity tensor. For one site type with hops $j$ of vector $\Delta\mathbf{r}_j$
    \[ D_{\alpha\beta} = \frac{1}{2}\sum_j \Gamma_j\, \Delta r_{j,\alpha}\, \Delta r_{j,\beta}. \]
    With two site types (4c, 8d), the master equation on the full hop network\cit{Smolders 2024}.
  \end{itemize}
\end{frame}

  \raggedright
  \begin{itemize}
    \item \textbf{$+1$:} the two sites are degenerate, and SOC changes nothing. The Pb-6p conduction band is empty.
    \item \textbf{Neutral:} the extra electron enters the Pb-6p conduction band, spread over the cell (shallow donor, magnetisation 0). SOC shifts that band by about 1 eV, so the neutral site energies depend on the method.
    \item \textbf{Consequence:} $+1$ results are robust at PBEsol. Neutral results need SOC (and possibly a hybrid functional).
  \end{itemize}
\end{frame}

\begin{frame}{Points for discussion}
  \small
  \begin{enumerate}
    \item \textbf{Strained NEBs.} Four fast hops (S2, S4, S1, S5) at four strains make 16 NEBs of 2--4 days on 4 nodes each. S2 and S4 alone carry most of the transport.
    \item \textbf{Charge state.} $V_\mathrm{I}$ is $+1$ throughout the gap (HSE06 + SOC), so the charge-state question is answered without a neutral NEB.
    \item \textbf{Phase result.} Strain does not stabilise $\gamma$ against $\delta$ at $\pm$1\,\%; both limits reported, with the kinetic reading.
    \item \textbf{Contrast with Steele \emph{et al.} (2019).} Their DFT found strain destabilised $\delta$ strongly against $\gamma$. How $\delta$ is strained (its own lattice here, the film interface there) needs a check.
    \item \textbf{Prefactors.} Kept at $10^{12}$ Hz; the axis order of the tensor stays open.
    \item \textbf{Fermi level.} A self-consistent $E_F$ needs acceptors ($V_\mathrm{Pb}$, $V_\mathrm{Cs}$), outside the present scope.
  \end{enumerate}
\end{frame}

\begin{frame}{Summary}
  \small
  \begin{itemize}
    \item \textbf{$V_\mathrm{I}$ is $+1$ for every Fermi level in the gap} (HSE06 + SOC edges). Its energies and barriers are robust at PBEsol.
    \item \textbf{Four short hops within 0.02 eV} (0.326--0.348 eV) carry the transport, each axis on a different hop. The tensor is nearly isotropic.
    \item \textbf{Strain does not stabilise $\gamma$}: the $\gamma$--$\delta$ gap (79 meV/f.u.) is smallest at zero strain and grows by up to 5 meV at $\pm$1\,\%. Clamping must act kinetically.
    \item \textbf{Tension lowers the vacancy formation energy} by about 0.043 eV per \%, while compression beyond $-0.5$\,\% changes little. Tensile films hold more iodine vacancies.
  \end{itemize}
  \vspace{0.6em}
  \centering
  \textbf{Next: strained NEBs (does tension also lower the barriers?) and the Freysoldt correction.}
\end{frame}
